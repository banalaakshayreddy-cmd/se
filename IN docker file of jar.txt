IN docker file of jar
//
FROM eclipse-temurin:17-jdk
COPY target/24bd1a6645-maven-project-0.0.1-SNAPSHOT.jar app.jar
ENTRYPOINT ["java", "-jar", "app.jar"]
//

docker build -t my-maven-app .

docker run --name my-maven-container my-maven-app

docker login

docker tag my-maven-app YOUR_USERNAME/my-maven-app:latest

docker push YOUR_USERNAME/my-maven-app:latest

//
FROM tomcat:9.0

EXPOSE 8080
//

docker build -t my-maven-webapp .
